Convenciones de Nombres en GDScript

Regla de oro: GDScript usa snake_case para casi todo. Solo las clases y tipos van en PascalCase. Las constantes en MAYUSCULAS_CON_GUIONES.
Tipo Estilo Ejemplo Tag
Variables snake_case velocidad_maxima, vida_actual var
Funciones snake_case recibir_daño(), esta_vivo() func
Constantes UPPER_SNAKE_CASE VELOCIDAD_MAX, VIDA_MAX const
Clases / Enums PascalCase class_name Jugador, enum Estado class
Señales snake_case (pasado) vida_cambiada, objeto_recogido signal
Privadas _snake_case _esta_muerto, _morir() _var
Señales conectadas _on_Nodo_señal _on_timer_timeout() _on_
Nodos @onready nombre descriptivo animacion, sprite, colision @onready
Tip nombres descriptivos: Prefiere velocidad_movimiento sobre vel, y puede_saltar sobre salto_bool. Los booleans con es_, puede_, tiene_, esta_.

📄 NombreScript.gd 
extends CharacterBody2D # o Node, Area2D, RigidBody2D...

# ─────────────────────────────────────────────
# CONSTANTES
# ─────────────────────────────────────────────
const VELOCIDAD_MAX: float = 300.0
const SALTO_FUERZA: float = -500.0
const VIDA_MAX: int = 100

# ─────────────────────────────────────────────
# SEÑALES
# ─────────────────────────────────────────────
signal vida_cambiada(vida_nueva: int)
signal personaje_muerto()

# ─────────────────────────────────────────────
# ENUMS
# ─────────────────────────────────────────────
enum Estado { IDLE, CORRIENDO, SALTANDO, MUERTO }

# ─────────────────────────────────────────────
# EXPORT (visibles en Inspector)
# ─────────────────────────────────────────────
@export var velocidad: float = 200.0
@export var vida: int = 100
@export var puede_correr: bool = true

# ─────────────────────────────────────────────
# ONREADY (nodos hijos)
# ─────────────────────────────────────────────
@onready var sprite: Sprite2D = $Sprite2D
@onready var animacion: AnimationPlayer = $AnimationPlayer
@onready var colision: CollisionShape2D = $CollisionShape2D

# ─────────────────────────────────────────────
# VARIABLES PRIVADAS
# ─────────────────────────────────────────────
var _esta_muerto: bool = false
var _estado_actual: Estado = Estado.IDLE
var _direccion: Vector2 = Vector2.ZERO

# ─────────────────────────────────────────────
# CICLO DE VIDA
# ─────────────────────────────────────────────
func _ready() -> void:
    _inicializar()

func _physics_process(delta: float) -> void:
    _leer_input()
    _mover(delta)

# ─────────────────────────────────────────────
# FUNCIONES PRIVADAS
# ─────────────────────────────────────────────
func _inicializar() -> void:
    vida = VIDA_MAX

func _leer_input() -> void:
    _direccion = Input.get_vector(
        "mover_izquierda", "mover_derecha",
        "mover_arriba", "mover_abajo")

func _mover(delta: float) -> void:
    velocity = _direccion * velocidad
    move_and_slide()

func _morir() -> void:
    _esta_muerto = true
    personaje_muerto.emit()

# ─────────────────────────────────────────────
# FUNCIONES PÚBLICAS
# ─────────────────────────────────────────────
func recibir_daño(cantidad: int) -> void:
    if _esta_muerto: return
    vida -= cantidad
    vida_cambiada.emit(vida)
    if vida <= 0:
        _morir()

func curar(cantidad: int) -> void:
    vida = clamp(vida + cantidad, 0, VIDA_MAX)
    vida_cambiada.emit(vida)

func esta_vivo() -> bool:
    return !_esta_muerto

# ─────────────────────────────────────────────
# SEÑALES CONECTADAS
# ─────────────────────────────────────────────
func _on_area_2d_body_entered(body: Node2D) -> void:
    if body.is_in_group("enemigos"):
        recibir_daño(10)
